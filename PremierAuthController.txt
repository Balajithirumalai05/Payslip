                    }
                }

                if (CreateTaskPR == "Y")
                {
                    if (objnexttask.ScreenArgs.UserRole == "RN")

                                AddonCost = Convert.ToDecimal(Dr.Value<string>("AddON") ?? "0.00")
                            };
                            obj.TotalServiceCost = obj.RxCost + obj.MedicalServiceCost + obj.GeneticsCost + obj.AddonCost + obj.RefillRxCost;

                        }
                    }
                    if (Ds.Tables.Count > 1 && Ds.Tables[1].Rows.Count > 0)
                    {
                        DataTable Dt = Ds.Tables[1];
                        foreach (DataRow Dr in Dt.Rows)
                        {
                            result.Add(new Accumulator()
                            {
                                BalanceAccumulator = Dr.Value<decimal>("Accumulator Balance")
                            });
                        }
                    }
                }
            }

            {

            }

        /// <summary>
        /// Common method that actually sends the email + converts AttachmentPath to PA-FAQ.pdf
        /// </summary>
        private void SendAuthStatusEmail(ObjectBase obj, string htmlBody, string[] imageNames)
        {
            if (!string.Equals(obj.EmailValidation, "Y", StringComparison.OrdinalIgnoreCase)
                || string.IsNullOrWhiteSpace(obj.EmailTo))
                return;

            try
            {
                var mailMessage = new System.Net.Mail.MailMessage();
                mailMessage.From = new System.Net.Mail.MailAddress(obj.EmailFrom, "WIN Healthcare");

                foreach (var to in obj.EmailTo.Split(new[] { ';', ',' }, StringSplitOptions.RemoveEmptyEntries))
                    mailMessage.To.Add(to.Trim());

                if (!string.IsNullOrWhiteSpace(obj.EmailCc))
                {
                    foreach (var cc in obj.EmailCc.Split(new[] { ';', ',' }, StringSplitOptions.RemoveEmptyEntries))
                        mailMessage.CC.Add(cc.Trim());
                }

                if (!string.IsNullOrWhiteSpace(obj.EmailBcc))
                {
                    foreach (var bcc in obj.EmailBcc.Split(new[] { ';', ',' }, StringSplitOptions.RemoveEmptyEntries))
                        mailMessage.Bcc.Add(bcc.Trim());
                }

                mailMessage.Subject = obj.EmailSubject ?? "";
                mailMessage.IsBodyHtml = true;

                var avHtml = System.Net.Mail.AlternateView.CreateAlternateViewFromString(
                    htmlBody, null, System.Net.Mime.MediaTypeNames.Text.Html);

                string imageFolder = @"\\winh\developerpaths\INDIA TEAM\Emailtemplates_test\";

                foreach (string fileName in imageNames)
                {
                    string fullPath = System.IO.Path.Combine(imageFolder, fileName);
                    if (System.IO.File.Exists(fullPath))
                    {
                        string ext = System.IO.Path.GetExtension(fullPath)?.ToLowerInvariant() ?? "";
                        string mediaType = ext == ".png" ? "image/png" :
                                           ext == ".gif" ? "image/gif" :
                                           System.Net.Mime.MediaTypeNames.Image.Jpeg;

                        var lr = new System.Net.Mail.LinkedResource(fullPath, mediaType);
                        lr.ContentId = fileName;
                        lr.TransferEncoding = System.Net.Mime.TransferEncoding.Base64;
                        avHtml.LinkedResources.Add(lr);
                    }
                }
                mailMessage.AlternateViews.Add(avHtml);

                // Attach AttachmentPath (Word docs auto-converted to PDF) named PA-FAQ.pdf
                if (!string.IsNullOrWhiteSpace(obj.AttachmentPath) && System.IO.File.Exists(obj.AttachmentPath))
                {
                    try
                    {
                        string pdfSourcePath = obj.AttachmentPath;
                        string ext = System.IO.Path.GetExtension(obj.AttachmentPath)?.ToLowerInvariant() ?? "";

                        if (ext == ".doc" || ext == ".docx")
                        {
                            try
                            {
                                string Temp = string.Empty;
                                string[] InteropPDFdllServers = Convert.ToString(
                                    System.Configuration.ConfigurationManager.AppSettings["InteropPDFdll_Servers"] ?? "").Split(',');

                                if (InteropPDFdllServers.Contains(System.Net.Dns.GetHostName().ToString().ToUpper()))
                                {
                                    Temp = Utilities.ConvertDocument.ConverttoPDF("", obj.AttachmentPath);
                                }
                                else
                                {
                                    Temp = Utilities.ConvertDocument.WordToPDF(obj.AttachmentPath);
                                }

                                if (!string.IsNullOrEmpty(Temp))
                                {
                                    int idx = Temp.LastIndexOf(",");
                                    string convertedPath = idx > 0 ? Temp.Substring(0, idx) : Temp;
                                    if (!string.IsNullOrWhiteSpace(convertedPath) && System.IO.File.Exists(convertedPath))
                                    {
                                        pdfSourcePath = convertedPath;
                                    }
                                }
                            }
                            catch (Exception exConv)
                            {
                                System.Diagnostics.Debug.WriteLine("PA-FAQ word-to-pdf conversion failed: " + exConv.Message);
                            }
                        }

                        var att = new System.Net.Mail.Attachment(pdfSourcePath);
                        att.Name = "PA-FAQ.pdf";
                        att.ContentType = new System.Net.Mime.ContentType("application/pdf");
                        mailMessage.Attachments.Add(att);
                    }
                    catch (Exception ex)
                    {
                        System.Diagnostics.Debug.WriteLine("PA-FAQ attachment failed: " + ex.Message);
                    }
                }

                // Extra attachment path (if exists)
                string extraPath = obj.Additional ?? "";
                if (!string.IsNullOrWhiteSpace(extraPath) && System.IO.File.Exists(extraPath))
                {
                    try
                    {
                        var att = new System.Net.Mail.Attachment(extraPath);
                        att.Name = System.IO.Path.GetFileName(extraPath);
                        mailMessage.Attachments.Add(att);
                    }
                    catch { }
                }

                using (var smtp = new System.Net.Mail.SmtpClient("10.10.20.108"))
                {
                    smtp.Send(mailMessage);
                }

                // Flag so the client-side VM does not send a duplicate email
                obj.EmailSent = true;
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine("SendAuthStatusEmail failed: " + ex.ToString());
            }
        }

        public ObjectBase UpdateMemberDetailsVisibility()
        {
            ObjectBase ObjContractInfo = new ObjectBase();
            try
            {
                RequestBase objReq = ATSParams?.ToJsonObject<RequestBase>();

                Dictionary<string, object> args = new Dictionary<string, object>();
                args.Add("@Contract", objReq.Contract ?? "");
                args.Add("@MemberID", objReq.MemberID ?? "");
                args.Add("@UserName", objReq.UserName ?? "");
                using (DataSet ds = ExecuteDB(SP_UpdateMemberDetailsVisibility, "", args))
                {
                    if (ds.Tables.Count > 0)
                    {
                        foreach (DataRow dr in ds.Tables[0].Rows)
                        {
                            ObjContractInfo = new ObjectBase()
                            {
                                Tag = dr.Value<string>("IsVisibleMemberDetails") ?? ""
                            };
                        }
                    }
                }

            }
            catch { }
            return ObjContractInfo;
        }

        private static string HtmlToPlainText(string html)
        {
            if (string.IsNullOrEmpty(html)) return string.Empty;
            try
            {
                // Drop script/style blocks entirely
                string text = Regex.Replace(html, @"<(script|style)[^>]*>[\s\S]*?</\1>", " ", RegexOptions.IgnoreCase);
                // Convert common block/line-break tags to newlines
                text = Regex.Replace(text, @"<\s*br\s*/?\s*>", "\n", RegexOptions.IgnoreCase);
                text = Regex.Replace(text, @"</\s*(p|div|tr|td|table|li|h[1-6])\s*>", "\n", RegexOptions.IgnoreCase);
                // Strip remaining tags
                text = Regex.Replace(text, "<[^>]+>", string.Empty);
                // Decode HTML entities
                text = System.Net.WebUtility.HtmlDecode(text);
                // Collapse whitespace
                text = Regex.Replace(text, @"[ \t]+", " ");
                text = Regex.Replace(text, @"\s*\n\s*", "\n");
                text = Regex.Replace(text, @"\n{3,}", "\n\n");
                return text.Trim();
            }
            catch
            {
                return Regex.Replace(html, "<[^>]+>", string.Empty).Trim();
            }
        }
