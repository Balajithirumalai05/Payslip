using WinDTO.Tasks;

namespace DynamicATS.Model
{
    public class AuditApprovalTaskModel : TaskBaseModel<AuditApprovedPrescriptionDetails>
    {
        public AuditApprovalTaskModel() :
        base("AuditApprovedPrescription/Save", "AuditApprovedPrescription/GetData")
        { }

        public System.Collections.Generic.List<string> GetFaxEnabledPharmacies(string contract)
        {
            return base.Get<System.Collections.Generic.List<string>>($"AuditApprovedPrescription/GetFaxEnabledPharmacies?contract={contract}");
        }
    }
}
