        [HttpPost]

        {

        [HttpPost]
        [ValidateFilter, ExceptionFilter]
        public string InsertAttachments(Clomid objMedicalAuthDetails, string TaskId)
        {
            try
            {
                Dictionary<string, object> _args = new Dictionary<string, object>();
                XDocument attchitems = new XDocument(new XElement("AttachItems"));
                // byte[] pdfbytes = null;
                if (objMedicalAuthDetails.NextTask.Attachments != null)
                {
                    foreach (var attachItem in objMedicalAuthDetails.NextTask.Attachments)
                    {
                        //if(attachItem!=null)
                        //{
                        //    pdfbytes = System.Convert.FromBase64String(attachItem.Additional);
                        //}
                        attchitems.Root.Add(new XElement("Item",
                            new XAttribute("FaxId", attachItem.ID ?? "0"),
                            new XAttribute("AttachmentPath", attachItem.Tag ?? "")
                            //   new XAttribute("Bytes", pdfbytes)
                            )
                            );
                    }
                    _args.Add("@TaskId", TaskId ?? "");
                    _args.Add("@AttachItems", attchitems.ToString());
                    // string result = string.Empty;
                    //result = 
                    ExecuteNonQuery(SP_InsertAttachments, objMedicalAuthDetails?.ScreenArgs?.Contract ?? "", _args);
                    return "";
                }
                else
                {
                    return "";
                }
            }
            catch (Exception ex)
            {
                return "";
            }
        }

        /* --- Save Method --- */
        [HttpPost]
        [ValidateFilter, ExceptionFilter]

        public object Save()
        {
            Clomid objClomid = ATSParams?.ToJsonObject<Clomid>();
            Dictionary<string, object> args = new Dictionary<string, object>();

            bool isCsa = string.Equals(
    objClomid?.ScreenArgs?.UserRole,
    "CSA",
    StringComparison.OrdinalIgnoreCase);

            bool isRn = string.Equals(
                objClomid?.ScreenArgs?.UserRole,
                "RN",
                StringComparison.OrdinalIgnoreCase);

            bool isSnr = string.Equals(
                objClomid?.ServiceData?.AdjudicatedService,
                "SNR",
                StringComparison.OrdinalIgnoreCase);

            /* --- Get Latest OutPatientID -- */
            string CancelReason = string.Empty;
            string SelectedCptcode = string.Empty;
            string LatestOutPatientID = string.Empty;

            bool isCsaIntermediateSnr =
     isCsa &&
     isSnr &&
     (string.Equals(objClomid?.ServiceData?.OutcomeStatus, "CSA Data Entered",
         StringComparison.OrdinalIgnoreCase)
      ||
      string.Equals(objClomid?.ServiceData?.OutcomeStatus, "CSA Data Reviewed",
         StringComparison.OrdinalIgnoreCase));

            try
            {
                if (isSnr && isRn)
                {
                    CancelReason = "Services not rendered";
                    SelectedCptcode = "SNR";
                }
                else if (objClomid.ServiceData.AdjudicatedService == "S4017-TF")
                {
                    CancelReason = "Cancelled at baseline";
                    SelectedCptcode = "ITCS";
                }
                else if (objClomid.ServiceData.AdjudicatedService == "S4042")
                {
                    CancelReason = "Cancelled after medication";
                    SelectedCptcode = "CMDCA";
                }
                else if (objClomid.ServiceData.AdjudicatedService == "S4042-TG")
                {
                    CancelReason = "Downcoded to CMDCL from CMDCLIUI";
                    SelectedCptcode = "CMDCL";
                }
                else if (objClomid.ServiceData.AdjudicatedService == "S4035-TF")
                {
                    CancelReason = "Upcoded to CMDCL With IUINS";
                    SelectedCptcode = "CMDCLIUI";
                }
                else
                {
                    CancelReason = "";
                    SelectedCptcode = "";
                }
            }
            catch (Exception ex)
            {
            }
            OutcomeController objOutcomeController = new OutcomeController();
            if (isRn && isSnr &&
                string.Equals(
                    objClomid?.ServiceData?.OutcomeStatus,
                    "Nurse Certified",
                    StringComparison.OrdinalIgnoreCase))
            {
                LatestOutPatientID = objOutcomeController.GetLatestOutPatientId(objClomid?.ScreenArgs?.MemberID ?? "", objClomid?.ScreenArgs?.Contract ?? "",
                    objClomid?.TaskDetails?.Speciality ?? "", objClomid?.ServiceData?.InitialCPTCode ?? "", objClomid?.ScreenArgs?.AuthNumber ?? "",
                    objClomid?.ScreenArgs?.OutPatientID ?? "", SelectedCptcode ?? "", CancelReason ?? "",
                    objClomid?.ScreenArgs?.UserName ?? "", objClomid?.ServiceData?.OldOutPatientID ?? "");
            }
            /* ---  header section  --- */
            args.Add("@Mode", "");
            args.Add("@AuthNo", objClomid?.ScreenArgs?.AuthNumber ?? "", DBNull.Value);
            args.Add("@MemberID", objClomid?.ScreenArgs?.MemberID ?? "");
            args.Add("@TimeOfEntry", DateTime.Now.ToString() ?? "");
            args.Add("@Contract", objClomid?.ScreenArgs?.Contract ?? "");
            args.Add("@Specialty", objClomid?.TaskDetails?.Speciality ?? "");
            args.Add("@UserName", objClomid?.ScreenArgs?.UserName ?? "");
            args.Add("@ServiceID", objClomid?.ScreenArgs?.OutPatientID ?? "", DBNull.Value);
            args.Add("@OldId", objClomid?.ScreenArgs?.OutPatientID ?? "", DBNull.Value);
            args.Add("@SNRCovidReason", objClomid?.ScreenArgs?.SNRReason ?? "", DBNull.Value);
            if (objClomid.TaskDetails.Reasons != null)
            { args.Add("@ReasonString", string.Join(",", objClomid?.TaskDetails?.Reasons?.Select(n => n.ID ?? "").ToArray())); }
            else { args.Add("@ReasonString", ""); }
            args.Add("@CycleNum", 0);
            args.Add("@OutcomeReceivedDate", objClomid?.TaskDetails?.OutcomeStartDate ?? "");
            args.Add("@CycleType", objClomid?.ServiceData?.CPTCode ?? "");
            args.Add("@SNR", (isSnr && isRn) ? 1 : 0);
            args.Add("@DurationFromDate", objClomid?.TaskDetails?.DateFrom ?? "", DBNull.Value);
            args.Add("@DurationToDate", objClomid?.TaskDetails?.DateTo ?? "", DBNull.Value);
            args.Add("@OutcomeStatus", objClomid?.ServiceData?.OutcomeStatus ?? "", DBNull.Value);
            /* ---  inf Cycle Hx  --- */
            /* ---  Section 1  --- */
            args.Add("@CancellationReason", (objClomid.Cancellation.CancelReason == null ? "" : (objClomid?.Cancellation?.CancelReason.ID ?? "")));
            args.Add("@CancellationDescOther", objClomid?.Cancellation?.OtherCancelReason ?? "");
            /* ---  Section 2  --- */
            args.Add("@LMPStartDate", objClomid?.BaselineResults?.LMPStartDate ?? "", DBNull.Value);
            args.Add("@BaseLineDate", objClomid?.BaselineResults?.BaselineFSHE2Date ?? "", DBNull.Value);
