                {
                    int i = 0;
                    foreach (DataRow row in ds.Rows)
                    {
                        string key = row["ShowTaskColumn"]?.ToString() ?? "";
                        string value = row["ShowTask"]?.ToString()?.Trim().ToUpper();

                        // If NULL or empty, default to "N"
                        if (string.IsNullOrEmpty(value))
                            value = "N";

                        // Store in dictionary
                        visible.TaskVisibility[key] = value;

                        // Set Additional1, Additional2, etc.
                        string propName = "Additional" + (i + 1);
                        var prop = visible.GetType().GetProperty(propName);
                        if (prop != null && prop.CanWrite)
                        {
                            prop.SetValue(visible, value);
                        }

                        i++;
                    }
                }
            }
            catch (Exception ex) { }
            return visible;
        }
