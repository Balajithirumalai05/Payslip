        [HttpPost]
        [ValidateFilter, ExceptionFilter]
        public string SaveAttachments()
        {
            try
            {
                AttachmentsInputDetails objSaveAttachmentsInput = ATSParams?.ToJsonObject<AttachmentsInputDetails>();
                Dictionary<string, object> args = new Dictionary<string, object>();
                args.Add("@TaskId", objSaveAttachmentsInput.TaskID ?? "");
                args.Add("@OutId", objSaveAttachmentsInput.OutID ?? "");
                args.Add("@Attach1", objSaveAttachmentsInput.AttachmentPath1 ?? "");
                args.Add("@Attach2", objSaveAttachmentsInput.AttachmentPath2 ?? "");
                args.Add("@Attach3", objSaveAttachmentsInput.AttachmentPath3 ?? "");
                int result = ExecuteNonQuery(SP_InsertAttachments, objSaveAttachmentsInput.Contract, args);
                string status = APIResponseStatus.APIResponse(result);
                return status;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"An error occurred: {ex.Message}");
                return "An error occurred while saving attachments.";
            }
        }

        [HttpPost]
        [ValidateFilter, ExceptionFilter]
        public string InsertAttachments(Premier objMedicalAuthDetails, string TaskId)
        {
            try
            {
                Dictionary<string, object> _args = new Dictionary<string, object>();
                XDocument attchitems = new XDocument(new XElement("AttachItems"));
                if (objMedicalAuthDetails.NextTask.Attachments != null)
                {
                    foreach (var attachItem in objMedicalAuthDetails?.NextTask?.Attachments)
                    {
                        attchitems.Root.Add(new XElement("Item",
                            new XAttribute("FaxId", attachItem.ID ?? "0"),
                            new XAttribute("AttachmentPath", attachItem.Tag ?? "")
                            )
                        );
                    }

                    _args.Add("@TaskId", TaskId);
                    _args.Add("@AttachItems", attchitems.ToString());

                    ExecuteNonQuery(SP_InsertAttachments, objMedicalAuthDetails?.ScreenArgs?.Contract ?? "", _args);
                }
                return "Attachments inserted successfully.";
            }
            catch (Exception ex)
            {

                Console.WriteLine($"An error occurred: {ex.Message}");
                return "An error occurred while inserting attachments.";
            }
        }

        [HttpPost]
        [ValidateFilter, ExceptionFilter]
        public object SavePremierDiagonistic()
        {

            PremierDiagnostic objPremier = ATSParams?.ToJsonObject<PremierDiagnostic>();

            bool isCsa = string.Equals(
        objPremier?.ScreenArgs?.UserRole,
        "CSA",
        StringComparison.OrdinalIgnoreCase);

            bool isRn = string.Equals(
                objPremier?.ScreenArgs?.UserRole,
                "RN",
                StringComparison.OrdinalIgnoreCase);

            bool isSnr = string.Equals(
                objPremier?.ServiceData?.AdjudicatedService,
                "SNR",
                StringComparison.OrdinalIgnoreCase);
            Dictionary<string, object> args = new Dictionary<string, object>();


            /* --- Get Latest OutPatientID -- */
            string CancelReason = string.Empty;
            string SelectedCptcode = string.Empty;
            string LatestOutPatientID = string.Empty;
            var output = this.GetSelectedCptCodeDetails(objPremier.ScreenArgs.AuthNumber ?? "", objPremier.ScreenArgs.MemberID ?? "", objPremier.ScreenArgs.Contract ?? "",
                objPremier.ServiceData.AdjudicatedService ?? "", objPremier.ServiceData.CPTCode ?? "");

            /* --- CSA intermediate SNR: when a CSA user picks Cycle Type = SNR with
                   OutcomeStatus = "CSA Data Entered" or "CSA Data Reviewed", do NOT
                   finalize the auth as SNR. Route to Primary UM Queue as Approval so
                   RN can Certify Outcome; SNR persists only when RN certifies. --- */
            bool isCsaIntermediateSnr = (objPremier.ScreenArgs.UserRole == "CSA")
                && (string.Equals(objPremier.ServiceData.OutcomeStatus, "CSA Data Entered", StringComparison.OrdinalIgnoreCase)
                    || string.Equals(objPremier.ServiceData.OutcomeStatus, "CSA Data Reviewed", StringComparison.OrdinalIgnoreCase))
                && string.Equals(objPremier.ServiceData.AdjudicatedService, "SNR", StringComparison.OrdinalIgnoreCase);

            if (output != null)
            {
                if (objPremier.ServiceData.AdjudicatedService == "SNR" && !isCsaIntermediateSnr)
                {
                    SelectedCptcode = "SNR";
                    CancelReason = "SNR";
                }
                else
                {
                    SelectedCptcode = output.ID;
                    CancelReason = output.Text;
                }
            }
            OutcomeController objOutcomeController = new OutcomeController();
