        [NonAction]

        {
            PharmacyDetails objPharmacyDetails = new PharmacyDetails();
            try
            {
                Dictionary<string, object> args = new Dictionary<string, object>();
                args.Add("@AuthNo", AuthNo ?? "");
                args.Add("@contract", Contract ?? "");
                using (DataSet ds = ExecuteDB(SP_GETPHARMACY, Contract, args))
                {
                    if (ds != null && ds.Tables.Count > 0)
                    {
                        if (ds.Tables[0] != null && ds.Tables[0].Rows.Count > 0)
                        {

                            {
                                objPharmacyDetails.PreviousChosenPharmCode = dr.Value<string>("PreviousChosenPharmCode") ?? "";
