        {
            string status="";
            try
            {
                AttachmentsInputDetails objSaveAttachmentsInput = ATSParams?.ToJsonObject<AttachmentsInputDetails>();
                Dictionary<string, object> args = new Dictionary<string, object>();
                args.Add("@TaskId", objSaveAttachmentsInput?.TaskID ?? "");
                args.Add("@OutId", objSaveAttachmentsInput?.OutID ?? "");
                args.Add("@Attach1", objSaveAttachmentsInput?.AttachmentPath1 ?? "");
                args.Add("@Attach2", objSaveAttachmentsInput?.AttachmentPath2 ?? "");
                args.Add("@Attach3", objSaveAttachmentsInput?.AttachmentPath3 ?? "");
                int result = ExecuteNonQuery(SP_InsertAttachments, objSaveAttachmentsInput?.Contract ?? "", args);
                status = APIResponseStatus.APIResponse(result);
                
            }
            catch (Exception ex)
            {
                Console.WriteLine("An error occurred: " + ex.Message);
            }
            return status;
        }
        [HttpPost]
        [ValidateFilter, ExceptionFilter]
        public string InsertAttachments(FET objMedicalAuthDetails, string TaskId)
        {
            try
            {
                Dictionary<string, object> _args = new Dictionary<string, object>();
                XDocument attchitems = new XDocument(new XElement("AttachItems"));
                // byte[] pdfbytes = null;
                if (objMedicalAuthDetails.NextTask.Attachments != null)
                {
                    foreach (var attachItem in objMedicalAuthDetails?.NextTask?.Attachments)
                    {
                        //if(attachItem!=null)
                        //{
                        //    pdfbytes = System.Convert.FromBase64String(attachItem.Additional);
                        //}
                        attchitems.Root.Add(new XElement("Item",
                            new XAttribute("FaxId", attachItem.ID ?? "0"),
                            new XAttribute("AttachmentPath", attachItem.Tag ?? "")
                            //   new XAttribute("Bytes", pdfbytes)
                            )
                            );
                    }

                    _args.Add("@TaskId", TaskId);
                    _args.Add("@AttachItems", attchitems.ToString());
                    // string result = string.Empty;
                    //result = 
                    ExecuteNonQuery(SP_InsertAttachments, objMedicalAuthDetails?.ScreenArgs?.Contract ?? "", _args);
                    return "";
                }
                else
                {
                    return "";
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("An error occurred: " + ex.Message);
                return "";
            }
        }

            try

                    {
                        objOutcomeController.SaveRx(objFET?.ScreenArgs?.AuthNumber ?? "", objFET?.ScreenArgs?.MemberID ?? "", "Progesterone",
                            objFET?.ProgesteroneDosage ?? "", objFET?.ProgesteroneDosage ?? "", objFET?.MedicationStartDate ?? "", objFET?.MedicationCancelDate ?? "",
                            "", "", objFET?.ProgesteroneDosageDays ?? "", "", objFET?.ScreenArgs?.Contract ?? "");
                    }
                    /*  --- Estradiol Injectable, Oral, Patch, Vaginal ---  */
                    if (string.IsNullOrWhiteSpace(objFET.EstradiolDosage) == false)
                    {
                        objOutcomeController.SaveRx(objFET.ScreenArgs?.AuthNumber ?? "", objFET.ScreenArgs?.MemberID ?? "", "Estradiol",
                            "", "", objFET?.MedicationStartDate ?? "", objFET?.MedicationCancelDate ?? "",
                            "", objFET?.EstradiolDosage ?? "", "", "", objFET?.ScreenArgs?.Contract ?? "");
                    }
                    if (string.IsNullOrWhiteSpace(objFET.ProgesteroneDosageOther) == false)
                    {
                        objOutcomeController.SaveRx(objFET.ScreenArgs?.AuthNumber ?? "", objFET?.ScreenArgs?.MemberID ?? "", objFET?.ProgesteroneDosageOther ?? "",
                            "", "", objFET?.MedicationStartDate ?? "", objFET?.MedicationCancelDate ?? "",
                            objFET?.ProgesteroneDosageSignature ?? "", "", "", "", objFET?.ScreenArgs?.Contract ?? "");
                    }
                }
                catch { }
                /* --- Save Outcome Status -- */
                string AdjudicatedCost = string.Empty;
                if (objFET.ServiceData.OutcomeStatus == "Missing Information")
                {
                    AdjudicatedCost = "0";
                }
                else
                {
                    AdjudicatedCost = "1";
                }
                objOutcomeController.SaveStatus(objFET?.ScreenArgs?.MemberID ?? "", objFET?.ScreenArgs?.AuthNumber ?? "",
                    objFET?.ServiceData?.OutcomeStatus ?? "", objFET?.ServiceData?.OutcomeStatus ?? "",
                    objFET?.ServiceData?.AdjudicatedService ?? "", AdjudicatedCost, objFET?.ScreenArgs?.UserName ?? "", objFET?.ScreenArgs?.Contract ?? "");
                ///* --- Save Outcome Received -- */
                //objOutcomeController.SaveReceived(objFET.ScreenArgs.Contract, objFET.ScreenArgs.UserName, objFET.ScreenArgs.AuthNumber,
                //    objFET.ScreenArgs.MemberID, objFET.ServiceData.AdjudicatedService, objFET.TaskDetails.Speciality);
                /* --- Move Pended Task -- */
                objOutcomeController.MovePendedTask("TaskMove", objFET?.ScreenArgs?.MemberID ?? "", objFET?.ScreenArgs?.AuthNumber ?? "", objFET?.ScreenArgs?.TaskID ?? "",
                    objFET?.ScreenArgs?.UserName ?? "", objFET?.ScreenArgs?.Contract ?? "");
                /* --- Edit Task -- */
                TaskController objTaskController = new TaskController();
                try
                {
                    if ((objFET.ScreenArgs.UserRole != "RN") && SelectedCptcode == "SNR")
                    {
                        objTaskController.Complete("1", objFET?.ScreenArgs?.TaskID ?? "", objFET?.NextTask?.Notes ?? "", objFET?.ScreenArgs?.UserName ?? "",
                            objFET?.ScreenArgs?.Contract ?? "");
                    }
                    else if (objFET.ScreenArgs.TaskName == objFET.NextTask.Task.TaskName)
                    {
                        objTaskController.Edit(objFET?.ScreenArgs?.TaskID ?? "", objFET?.ScreenArgs?.UserName ?? "", objFET?.NextTask?.Notes ?? "",
                            objFET?.NextTask.Task.SendToQueue.Text ?? "", objFET.NextTask.Task.Priority ?? "", DateTime.Now.ToString(), objFET.ScreenArgs.Contract);
                    }
                    else if (objFET.ScreenArgs.TaskName != objFET.NextTask.Task.TaskName)
                    {
                        /* --- Complete Task -- */
                        objTaskController.Complete("1", objFET?.ScreenArgs?.TaskID ?? "", objFET?.NextTask?.Notes ?? "", objFET?.ScreenArgs?.UserName ?? "",
                            objFET?.ScreenArgs?.Contract ?? "");
                        /* --- Create Task -- */
                        if (!string.IsNullOrWhiteSpace((objFET?.NextTask?.Task?.TaskName ?? "")))
                        {
                            objTaskController.Create(objFET?.ScreenArgs?.Contract ?? "", objFET?.TaskDetails?.Speciality ?? "", objFET?.ScreenArgs?.MemberID ?? "",
                                objFET?.ScreenArgs?.UserName ?? "", objFET?.NextTask.Task?.TaskName ?? "", objFET?.NextTask.Task.SendToQueue?.Text ?? "",
                                objFET?.ScreenArgs?.AuthNumber ?? "", "", objFET?.NextTask?.Notes ?? "", objFET?.NextTask.Task?.Priority ?? "", "");
                        }
                    }
                }
                catch { }
                if ((objFET.ScreenArgs.UserRole == "RN") && (objFET.ServiceData.OutcomeStatus == "Pending Clinical Review" || objFET.ServiceData.OutcomeStatus == "Nurse Certified"))
                {
                     sTaskId = objTaskController.Create(objFET?.ScreenArgs?.Contract ?? "", objFET?.TaskDetails?.Speciality ?? "", objFET?.ScreenArgs?.MemberID ?? "",
                         objFET?.ScreenArgs?.UserName ?? "", "Certify Outcome (Data)", objFET?.NextTask.Task.SendToQueue?.Text ?? "",
                         objFET?.ScreenArgs?.AuthNumber ?? "", "", objFET?.NextTask?.Notes ?? "", objFET?.NextTask.Task?.Priority ?? "", "");
                    objTaskController.Complete("1", sTaskId, objFET?.NextTask?.Notes ?? "", objFET?.ScreenArgs?.UserName ?? "",
                       objFET?.ScreenArgs?.Contract ?? "");
                }
                /* --- update work flow status -- */
                string Reasons = string.Empty;
                if (objFET.TaskDetails.Reasons != null)
                { Reasons = string.Join(",", objFET?.TaskDetails?.Reasons?.Select(n => n.ID ?? "").ToArray()); }
                if (objFET.ScreenArgs.UserRole == "CSA") { objFET.TaskDetails.WorkflowStep = "Outcome Setup"; }
                else { objFET.TaskDetails.WorkflowStep = "Outcome Review"; }
                if (string.IsNullOrEmpty(LatestOutPatientID)) { LatestOutPatientID = objFET?.ScreenArgs?.OutPatientID ?? "1"; }
                objOutcomeController.UpdateWorkFlowStatus(objFET.ScreenArgs?.AuthNumber ?? "", objFET.TaskDetails?.WorkflowStep ?? "",
                    objFET?.ServiceData?.OutcomeStatus ?? "", Reasons, LatestOutPatientID, objFET?.ScreenArgs?.UserName ?? "", objFET?.ScreenArgs?.Contract ?? "",
                    objFET?.ScreenArgs?.TaskName ?? "", objFET?.ServiceData?.OutcomeStatus ?? "", objFET?.ScreenArgs?.UserRole ?? "", objFET?.NextTask.Task?.TaskName ?? "", SelectedCptcode);

                /* --- Save Outcome Received -- */
                objOutcomeController.SaveReceived(objFET.ScreenArgs?.Contract ?? "", objFET.ScreenArgs?.UserName ?? "", objFET.ScreenArgs?.AuthNumber ?? "",
                    objFET?.ScreenArgs?.MemberID ?? "", objFET?.ServiceData?.AdjudicatedService ?? "", objFET?.TaskDetails?.Speciality ?? "");
                InsertAttachments(objFET, objFET?.ScreenArgs?.TaskID ?? "");
                try
                {
                    if (objFET?.ServiceData?.OutcomeStatus == "Nurse Certified")
                    {
                        objFET.NextTask.Notes = "";
                        objTaskController.Complete("1", sTaskId ?? "", objFET.NextTask.Notes ?? "", objFET.ScreenArgs?.UserName ?? "",
         objFET?.ScreenArgs?.Contract ?? "");
                    }
                }
                catch(Exception ex) { }
            }
            catch (Exception ex)
            {
                Console.WriteLine("An error occurred: " + ex.Message);
            }
            return new { ID = LatestOutPatientID };
        }

        #endregion
    }
