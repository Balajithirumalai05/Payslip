        #region Command

        private void HandleCommand(object args)

        {
            try
            {
                if (this.OnPreExecuteCommand(args))
                {
                    if (args != null)
                    {
                        if (args is string)
                        {
                            string command = args.ToString().ToUpper().Trim();
                            if (string.IsNullOrWhiteSpace(command) == false)
                            {
                                switch (command)
                                {
                                    case CMD_MSG_SAVE:
                                        this.Save();
                                        break;
                                    case CMD_MSG_CLOSE:
                                        //int Result = this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Confirmation, "Changes will not be saved. Do you wish to continue? ", new string[] { "Yes", "No" });
                                        //if (Result == 0)
                                        //{
                                        //    this.Close();
                                        //}
                                        //else if (Result == 1)
                                        //{
                                        //    return;
                                        //}
                                        this.Callback?.Invoke(new ViewModelMsg() { Message = AppUtils.MSG_CLOSE_SECTION });
                                        OnClosing();
                                        break;
                                    case CMD_MSG_EXPAND:
                                        this.View.Send(AppUtils.MSG_EXPAND_SECTIONS);
                                        break;
                                    case CMD_MSG_COLLAPSE:
                                        this.View.Send(AppUtils.MSG_COLLAPSE_SECTIONS);
                                        break;
                                    case CMD_MSG_CHECKLIST:
                                        if (this.Window != null && this.Window is IWindowMsg)
                                        {
                                            (this.Window as IWindowMsg).Send(MemberSnapshotVM.MSG_OPEN_CHECKLIST, CheckListDetails);
                                        }
                                        break;
                                }
                            }
                        }
                        else if (args is ObjectBase)
                        { this.OpenDoc(args as ObjectBase); }
                    }
                    if ((OutPatientIdResult ?? "") != "")
                    {
                        this.OnPostExecuteCommand(args);
                    }
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("An exception captured " + ex.Message);
            }
        }

        private void Save()
        {
            try
            {
                if (this.OnValidateInputs() && this.OutcomeHeaderValidation() && this.OnDateValidate() && CheckLock(this.Data))
                {
                    if (this.Data != null && this.Data is OutcomeBase)
                    {
                        OutcomeBase data = this.Data as OutcomeBase;


                        if ((data?.ScreenArgs?.TaskName ?? "") == "Long Term Outcome")
                        {
                            OnCheckMissingFields();
                        }
                        else
                        {
                            if (data.ServiceData == null) { data.ServiceData = new OutcomeServiceData(); }

                            /*----- CSA selecting SNR: AdjudicatedService is still set to "SNR"
                                    via the normal path below so that when RN reopens the Certify
                                    Outcome task, Cycle Type correctly pre-selects "SNR" (it is
                                    derived from AdjudicatedService in Init()). Finalization (auth
                                    status flip to SNR, denied-service row creation) is prevented
                                    server-side via role checks in PremierOutcomeController /
                                    OutcomeController, so the auth remains "Approval" until the RN
                                    actually certifies with SNR. -----*/
                            if (this.SelectedCycleType != null && string.IsNullOrWhiteSpace(this.SelectedCycleType.ID) == false)
                            {
                                data.ServiceData.AdjudicatedService = AppUtils.GetCycleTypeUniqueID(this.SelectedCycleType.ID);
                                data.ServiceData.MainSection = AppUtils.GetCycleTypeMainSection(this.SelectedCycleType.ID);
                            }
                            /*----- Missing Fields Popup -----*/
                            bool Certify = false;
                            bool OnlyBlank = false;
                            List<SectionMissingFieldInfo> sections = this.OnCheckMissingFields();
                            string blankSections = string.Empty;
                            try { blankSections = this.View.Send(AppUtils.MSG_BLANK_SECTIONS) as string; } catch { }
                            if (string.IsNullOrWhiteSpace(blankSections) == false && sections.Count > 0)
                            {
                                List<string> blankSectionslist = blankSections.Split(',').ToList();

                                List<string> sectionslist = new List<string>();
                                foreach (var section in sections)
                                    sectionslist.Add(section.Header);
                                IEnumerable<string> Common = blankSectionslist.Except(sectionslist);
                                string MissingBlankSection = string.Empty;
                                foreach (var c in Common)
                                {
                                    if (string.IsNullOrWhiteSpace(MissingBlankSection) == false) { MissingBlankSection += ", "; }
                                    MissingBlankSection += c;
                                }
                                if (string.IsNullOrWhiteSpace(MissingBlankSection) == false)
                                {
                                    var Mblank = new SectionMissingFieldInfo() { Header = MissingBlankSection, Fields = new List<string>() { "Please enter atleast one field" } };
                                    sections.Add(Mblank);
                                }
                                var item = new SectionMissingFieldInfo() { Header = "Blank Section(s)", Fields = new List<string>() { blankSections } };
                                if (sections.Count == 0) { sections.Add(item); OnlyBlank = true; }
                                else { sections.Insert(0, item); }
                            }
                            if (sections.Count > 0)
                            {
                                bool queue = false;
                                if (string.Compare(AppUtils.USER_ROLE_RN, Utilities.Common.Role ?? "", true) == 0)
                                { if (OnlyBlank) { queue = true; } }

                                if (data.ServiceData.OutcomeStatus == "Incomplete" || data.ServiceData.OutcomeStatus == "Missing Information" ||
                                    data.ServiceData.OutcomeStatus == "Provider Data Entered")
                                {
                                    data.NextTask.Task.SendToQueue = new ObjectBase();
                                    data.NextTask.Task.SendToQueue.Text = "My Personal Queue";
                                }
                                else if (data.ServiceData.OutcomeStatus == "Pending Clinical Review")
                                {
                                    data.NextTask.Task.SendToQueue.Text = "Pending Clinical Review";
                                }
                                else if (data.ServiceData.OutcomeStatus == "Nurse Certified")
                                {
                                    data.NextTask.Task.SendToQueue.Text = "Pending Clinical Review";
                                    data.ServiceData.OutcomeStatus = "Pending Clinical Review";
                                }
                                else if (data.ServiceData.OutcomeStatus == "CSA Data Entered" ||
                                    data.ServiceData.OutcomeStatus == "CSA Data Reviewed")
                                {
                                    data.NextTask.Task.SendToQueue.Text = "Primary UM Queue";
                                }
                                else
                                {
                                    data.NextTask.Task.SendToQueue = null;
                                }



                                MissingFieldPopupVM datacontext = new MissingFieldPopupVM()
                                {
                                    MissingReasonList = this.AuthReasonList,
                                    SelectedAuthReasonList = data.TaskDetails.Reasons,
                                    SectionList = sections,
                                    QueueList = WorkQueueList,
                                    RNQueue = queue,
                                    SelectQueue = data.NextTask.Task.SendToQueue//,
                                    //MemberID = this.MemberId,
                                    //SectionName = this.Screen,
                                    //Contract = this.Contract,
                                };
                                this.View.Send(AppUtils.MSG_MISSING_FIELD, datacontext);

                                if (datacontext.Status == 0)
                                { Certify = true; data.ServiceData.OutcomeStatus = "Nurse Certified"; }
                                else if (datacontext.Status == 2)
                                { return; }
                                else if (datacontext.Status == 1)
                                { data.NextTask.Task.SendToQueue.Text = datacontext.SelectQueue.Text; }

                            }
                            else if (sections.Count == 0)
                            {
                                if ((data?.ScreenArgs?.UserRole ?? "") == "CSA")
                                {
                                    //if ((data?.ScreenArgs?.TaskName ?? "") == "Enter Outcome") { data.ServiceData.OutcomeStatus = "CSA Data Entered"; }
                                    if ((data?.ScreenArgs?.TaskName ?? "") == "Review Online Outcome") { data.ServiceData.OutcomeStatus = "CSA Data Reviewed"; }
                                }

                                else
                                {
                                    if (data.ServiceData.OutcomeStatus == "Incomplete" || data.ServiceData.OutcomeStatus == "Missing Information" ||
                                   data.ServiceData.OutcomeStatus == "Pending Clinical Review")
                                    { Certify = false; }

                                    else
                                    {
                                        Certify = true;
                                        data.ServiceData.OutcomeStatus = "Nurse Certified";
                                    }



                                }
                            }
                            /*----- Nurse Certification Popup -----*/
                            if (Certify)
                            {
                                if (string.Compare(AppUtils.USER_ROLE_RN, Utilities.Common.Role ?? "", true) == 0)
                                {
                                    bool showNurseCertifyScreen = true;
                                    if (string.Compare((this._previousStatus ?? "").Trim(), AppUtils.ATS_STATUS_NURSE_CERTIFIED, true) == 0)
                                    {
                                        if (this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Confirmation, "Do you want to re-certify?", new string[] { "Yes", "No" }) != 0)
                                        { showNurseCertifyScreen = false; }
                                    }



                                    if (showNurseCertifyScreen == true)
                                    {
                                        string TempSystemDetermineService = string.Empty;
                                        ObjectBase TempPreviousService = new ObjectBase();
                                        string Temp = string.Empty;
                                        ///// string Embriocount = string.Empty;

                                        if (this.SelectedCycleType.Text.Contains("/")) { TempSystemDetermineService = (this.SelectedCycleType.Text.Split(new char[] { '/' })[1] ?? "").Trim(); }
                                        TempPreviousService = this.CycleTypeList.Find((x) => { return string.Compare(this._previousService ?? "", x.ID, true) == 0; });
                                        if (TempPreviousService != null)
                                        {
                                            if (TempPreviousService.Text.Contains("/")) { Temp = (TempPreviousService.Text.Split(new char[] { '/' })[1] ?? "").Trim(); }
                                        }
                                        if (data.ServiceData.InitialCPTCode.Equals("Global IVF"))
                                        {
                                            this.RenderAsPerInitiallyApproved = false;
                                        }
                                        else
                                        {
                                            this.RenderAsPerInitiallyApproved = true;
                                        }
                                        if (Data != null && Data is FET)
                                        {
                                            this.CPTCode = "FETA";
                                            this.OutcomeType = "fet";
                                        }
                                        else if (Data != null && Data is Clomid)
                                        {
                                            this.CPTCode = "Clomid";
                                            this.OutcomeType = "clo";
                                        }
                                        else if (Data != null && Data is Surgical)
                                        {
                                            this.CPTCode = "Surgical";
                                            this.OutcomeType = "sur";
                                        }
                                        else if (Data != null && Data is NaturalCycle)
                                        {
                                            this.CPTCode = "NaturalCycle";
                                            this.OutcomeType = "nc";
                                        }
                                        else if (Data != null && Data is IVFDonor)
                                        {
                                            this.CPTCode = "IVFDonar";
                                            this.OutcomeType = "ivfdon";
                                        }
                                        else if (Data != null && Data is IVF)
                                        {
                                            this.CPTCode = "IVF";
                                            this.OutcomeType = "ivf";
                                        }
                                        else if (Data != null && Data is Gonadotropin)
                                        {
                                            this.CPTCode = "Gonadotropin";
                                            this.OutcomeType = "gon";
                                        }
                                        else if (Data != null && Data is Diagnostic)
                                        {
                                            this.CPTCode = "Diagnostic";
                                            this.OutcomeType = "dia";
                                        }
                                        else if (Data != null && Data is MedicalTreatment)
                                        {
                                            this.CPTCode = "MEDTR";
                                            this.OutcomeType = "medt";
                                        }
                                        else if (Data != null && Data is Oocyte)
                                        {
                                            this.CPTCode = "Oocyte";
                                            this.OutcomeType = "oocyte";
                                        }
                                        else if (Data != null && Data is OCP)
                                        {
                                            this.CPTCode = "OCP";
                                            this.OutcomeType = "oocytecryo";
                                        }
                                        else if (Data != null && Data is Premier)
                                        {
                                            this.CPTCode = data.ServiceData.InitialCPTCode;
                                            this.OutcomeType = "Premier";
                                        }
                                        else if (Data != null && Data is PremierDiagnostic)
                                        {
                                            this.CPTCode = "Diagnostic";
                                            this.OutcomeType = "dia";
                                        }

                                        else if (Data != null && Data is Igenomix)
                                        {
                                            Igenomix igenomix = new Igenomix();
                                            igenomix = this.Data as Igenomix;
                                            this.CPTCode = this.SelectedCycleType.Text;
                                            this.OutcomeType = "Igenomix";
                                            this.IgenomixOutcomeType = "Y";
                                            this.Embriocount = igenomix.EmbryosRecieved;
                                            data.ServiceData.InitialCPTCode = this.SelectedCycleType.Text;
                                            TempSystemDetermineService = this.SelectedCycleType.Text;
                                            IgenomixIntialService = igenomix.InitialService;
                                        }


                                        RequestBase input = new RequestBase()
                                        {
                                            Contract = data.ScreenArgs.Contract ?? "",
                                            MemberID = data.ScreenArgs.MemberID ?? "",
                                            AuthNo = data.ScreenArgs.AuthNumber ?? "",
                                            FormattedAuth = data.ScreenArgs.FormattedAuthNumber ?? "",
                                        };
                                        var List = this.Model.GetOutcomeCycles(input);

                                        this.CompletedCycles = List.Where((x) => { return string.Compare("N", x.CancellationType, true) == 0; }).ToList<OutcomeProcedureCodes>();
                                        this.CancelledCycles = List.Where((x) => { return string.Compare("Y", x.CancellationType, true) == 0; }).ToList<OutcomeProcedureCodes>();

                                        NurseCertificationPopupVM datacontext = new NurseCertificationPopupVM()
                                        {
                                            UpcodeList = this.Model.GetCycleTypeUpcodes(this.CPTCode ?? "",
                                                                data.ServiceData.InitialCPTCode, data.ScreenArgs.Contract, this.CycleTypeList, data.ScreenArgs.MemberID ?? "",
                                                                data.ScreenArgs.AuthNumber ?? "", this.IgenomixOutcomeType, Embriocount),
                                            // UpcodeList = AppUtils.GetCycleTypeUpcodes(this.CycleTypeList, data.ServiceData.AdjudicatedService, data.ServiceData.MainSection),
                                            // DowncodeList = AppUtils.GetCycleTypeDowncodes(this.CycleTypeList, data.ServiceData.AdjudicatedService, data.ServiceData.MainSection),
                                            DowncodeList = this.Model.GetCycleTypeDowncodes(this.CPTCode ?? "",
                                                                    data.ServiceData.InitialCPTCode, data.ScreenArgs.Contract,
                                                                    this.CycleTypeList, data.ScreenArgs.MemberID ?? "",
                                                                    data.ScreenArgs.AuthNumber ?? "", this.IgenomixOutcomeType, Embriocount),

                                            InitiallyApprovedService = data.ServiceData.InitialCPTCode ?? "",
                                            PreviousCertifiedService = Temp,
                                            SystemDeterminedService = TempSystemDetermineService ?? "",

        private void OpenDoc(ObjectBase attachment)
        {
            try
            {
                if (string.IsNullOrWhiteSpace(attachment.Tag) == false && (attachment.Tag.Contains("\\") || attachment.Tag.Contains("/")))
                { try { Process.Start(attachment.Tag); } catch { this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Error, "Unable to open document, please contact IT."); } }
                else
                {
                    if (this.Data != null && this.Data is TaskBase)
                    {
                        TaskBase data = this.Data as TaskBase;

                        string tempdocpath = TempCoverFaxesPath;
                        string Targetpath = string.Empty;

                        bool exists = System.IO.Directory.Exists(tempdocpath + "\\" + Utilities.Common.UserFullName + "\\" + data.ScreenArgs.Contract + "\\" + "Temp");
                        if (exists)
                        {
                            Targetpath = tempdocpath + "\\" + Utilities.Common.UserFullName + "\\" + data.ScreenArgs.Contract + "\\" + "Temp";
                        }
                        else
                        {
                            System.IO.Directory.CreateDirectory(tempdocpath + "\\" + Utilities.Common.UserFullName + "\\" + data.ScreenArgs.Contract + "\\" + "Temp");
                            Targetpath = tempdocpath + "\\" + Utilities.Common.UserFullName + "\\" + data.ScreenArgs.Contract + "\\" + "Temp";
                        }

                        byte[] bytes = this.Model.GetDocBytes(new DocBytesParams() { Contract = data.ScreenArgs.Contract, DocID = attachment.ID });
                        if (bytes != null && bytes.Length > 0)
                        {
                            string filename = Targetpath;
                            if (filename.Contains("\\") && filename.EndsWith("\\") == false) { filename += "\\"; }
                            if (filename.Contains("/") && filename.EndsWith("/") == false) { filename += "/"; }
                            filename += Guid.NewGuid().ToString() + attachment.Text;// + ".pdf";
                            File.WriteAllBytes(filename, bytes);
                            Process.Start(filename);
                        }
                        else if (string.IsNullOrWhiteSpace(attachment.Additional) == false)
                        {
                            //string filename = AppDomain.CurrentDomain.BaseDirectory;
                            string filename = Targetpath;
                            if (filename.Contains("\\") && filename.EndsWith("\\") == false) { filename += "\\"; }
                            if (filename.Contains("/") && filename.EndsWith("/") == false) { filename += "/"; }
                            filename += Guid.NewGuid().ToString() + attachment.Text; //+ ".pdf";
                            File.WriteAllBytes(filename, Convert.FromBase64String(attachment.Additional));
                            Process.Start(filename);
                        }

        private bool CheckLock(T args)
        {
            bool isAvailable = false;
            try
            {
                var data = Data as ExTaskBase;
                LockMember lockedBy = new MemberSnapshotModel().ManageMemberLock(new LockMember()
                {
                    Contract = data?.ScreenArgs?.Contract ?? "",
                    UserName = Utilities.Common.UserName ?? "",
                    TaskId = "",
                    MemberId = data?.ScreenArgs?.MemberID ?? "",
                    ProcessType = "Lock Check"
                });
                isAvailable = (string.Compare(lockedBy.UserName, null) == 0);
                if (string.Equals(lockedBy?.UserName, Utilities.Common.UserFullName, StringComparison.OrdinalIgnoreCase))
                { isAvailable = true; }
                if (!isAvailable)
                {
                    if ((lockedBy?.UserName ?? "") == "Inactive Member")
                    {
                        // this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Information, "Inactive/Duplicate member profile case, Please contact ITProdSupport.");
                        isAvailable = true;
                    }
                    else
                    {
                        this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Information, "Member case is locked by " + (lockedBy?.UserName ?? ""));
                    }
                }
            }
            catch { isAvailable = true; }
            return isAvailable;
        }
        private void SelectedCyclePropertyChangedHandler(object sender, PropertyChangedEventArgs e)
        {
            try
            {
                if (this.Data != null && this.Data is OutcomeBase)
                {
                    OutcomeBase data = this.Data as OutcomeBase;
                    if (string.Compare(e.PropertyName, nameof(this.SelectedCycleType), true) == 0)
                    {
                        if (this.SelectedCycleType.Text == "SNR")
                        {
                            /* CSA intermediate SNR (CSA Data Entered / CSA Data Reviewed):
                               route to RN via Certify Outcome in Primary UM Queue. RN commits SNR on certify. */
                            if (!Utilities.Common.IsNurse
                                && (string.Equals(data.ServiceData.OutcomeStatus, "CSA Data Entered", StringComparison.OrdinalIgnoreCase)
                                    || string.Equals(data.ServiceData.OutcomeStatus, "CSA Data Reviewed", StringComparison.OrdinalIgnoreCase)))
                            {
                                if (data.NextTask == null) { data.NextTask = new NextTaskDetails(); }
                                if (data.NextTask.Task == null) { data.NextTask.Task = new WinDTO.Tasks.TaskInfo(); }
                                data.NextTask.Task.TaskName = "Certify Outcome";
                                data.NextTask.Task.SendToQueue = new ObjectBase() { Text = "Primary UM Queue" };
                                data.NextTask.Task.Priority = "Medium";
                            }
                            else
                            {
                                data.NextTask.Task.TaskName = "";
                                data.NextTask.Task.SendToQueue = new ObjectBase();
                                data.NextTask.Task.Priority = "";
                            }
                            if (Utilities.Common.IsNurse)
                            { data.ServiceData.OutcomeStatus = "Complete"; }

                        }
                        else
                        {
                            if (data.NextTask == null)
                            { data.NextTask = new NextTaskDetails(); }
                            data.NextTask.Task = this.Model.GetNextTask(new NextTaskParams
                            {
                                Contract = data.ScreenArgs.Contract ?? "",
                                UserRole = Utilities.Common.Role,
                                TaskName = data.ScreenArgs.TaskName ?? "",
                                TaskStatus = data.ServiceData.OutcomeStatus ?? ""
                            });
                        }
                        if (string.Equals(this.SelectedCycleType.LoginId, "Y", StringComparison.OrdinalIgnoreCase))
                        {
                            RequestBase request = new RequestBase()
                            {
                                Contract = data?.ScreenArgs?.Contract ?? "",
                                MemberID = data?.ScreenArgs?.MemberID ?? "",
                                AuthNo = data?.ScreenArgs?.AuthNumber ?? "",
                                AAPTaskid = data.ScreenArgs.TaskID,
                                OutPatientId = data?.AuthHeaderDetails?.OutpatientId ?? "",
                                AuthStatus = (data?.AuthHeaderDetails?.AuthStatus?.Text ?? ""),
                                UserName = data?.AuthHeaderDetails.LastUser ?? "",
                                WebLink = data?.TaskDetails?.ProcedureCode ?? ""
                            };
                            var result = this.View.Send(AppUtils.MSG_SNRDENIEL_REASON, request);
                            data.ScreenArgs.SNRReason = result?.ToString() ?? "";
                            if (!(string.IsNullOrWhiteSpace(data.ScreenArgs.SNRReason)))
                            {
                                data.NextTask.Notes += "Outcome SNRDenial Reason(s): " + Environment.NewLine + data.ScreenArgs.SNRReason + Environment.NewLine;
                            }

                        }
                    }

                }
                if (this.Data != null && this.Data is Igenomix)
                {
                    Igenomix data = this.Data as Igenomix;
                    if (string.Compare(e.PropertyName, nameof(this.SelectedCycleType), true) == 0)
                    {
                        IgenomixOutcomeVM._selectedCycle = SelectedCycleType;

                    }
                }
                if (this.Data != null && this.Data is Premier)
                {
                    Premier data = this.Data as Premier;
                    if (string.Compare(e.PropertyName, nameof(this.SelectedCycleType), true) == 0)
                    {
                        if (this.SelectedCycleType.ID == "IUI-2-INS-CYC")
                        {
                            data.IsVisibleSecondInseminationDate = true;
                        }
                        else
                        {
                            data.IsVisibleSecondInseminationDate = false;
                        }

                    }
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("An exception captured " + ex.Message);
            }
        }

                                if (!Utilities.Common.IsNurse
                                    && (string.Equals(data.ServiceData.OutcomeStatus, "CSA Data Entered", StringComparison.OrdinalIgnoreCase)
                                        || string.Equals(data.ServiceData.OutcomeStatus, "CSA Data Reviewed", StringComparison.OrdinalIgnoreCase)))
                                {
                                    if (data.NextTask == null) { data.NextTask = new NextTaskDetails(); }
                                    if (data.NextTask.Task == null) { data.NextTask.Task = new WinDTO.Tasks.TaskInfo(); }
                                    data.NextTask.Task.TaskName = "Certify Outcome";
                                    data.NextTask.Task.SendToQueue = new ObjectBase() { Text = "Primary UM Queue" };
                                    data.NextTask.Task.Priority = "Medium";
                                }
                                else
                                {
                                    data.NextTask.Task.TaskName = "";
                                    data.NextTask.Task.SendToQueue = new ObjectBase();
                                    data.NextTask.Task.Priority = "";
                                }

                            }
                            else
                            {
                                if (data.NextTask == null)
                                { data.NextTask = new NextTaskDetails(); }
                                string temp = string.Empty;

                                if (string.IsNullOrWhiteSpace(data?.NextTask?.Task?.SendToQueue?.Text ?? "") == false)
                                {
                                    temp = data.NextTask.Task.SendToQueue.Text ?? "";
                                }
                                if (data.NextTask.Task.SendToQueue.Text == "My Personal Queue" && data.ServiceData.OutcomeStatus == "CSA Data Entered")
                                {
                                    temp = string.Empty;
                                }
                                else
                                {
                                    data.NextTask.Task.SendToQueue = new ObjectBase() { Text = "" };
                                }
                                data.NextTask.Task = this.Model.GetNextTask(new NextTaskParams
                                {
                                    Contract = data.ScreenArgs.Contract ?? "",
                                    UserRole = Utilities.Common.Role,
                                    TaskName = data.ScreenArgs.TaskName ?? "",
                                    TaskStatus = data.ServiceData.OutcomeStatus ?? ""
                                });
                                if (temp != string.Empty)
                                {
                                    data.NextTask.Task.SendToQueue.Text = temp;
                                }
                                try
                                {
                                    if (data.ServiceData.OutcomeStatus == "Pending Clinical Review")
                                    {
                                        data.NextTask.Task.SendToQueue.Text = "Pending Clinical Review";
                                    }
                                }
                                catch (Exception ex)
                                {
                                    Console.WriteLine("An exception captured " + ex.Message);
                                }
                            }
                        }
                        else
                        {
                            if (data.NextTask == null)
                            { data.NextTask = new NextTaskDetails(); }
                            data.NextTask.Task = this.Model.GetNextTask(new NextTaskParams
                            {
                                Contract = data.ScreenArgs.Contract ?? "",
                                UserRole = Utilities.Common.Role,
                                TaskName = data.ScreenArgs.TaskName ?? "",
                                TaskStatus = data.ServiceData.OutcomeStatus ?? ""
                            });
                        }
                        switch (data.ServiceData.OutcomeStatus)
                        {
                            case AppUtils.ATS_STATUS_INCOMPLETE:
                            case AppUtils.ATS_STATUS_MISSING_INFO:
                            case AppUtils.ATS_STATUS_PROVIDER_DATA_ENTERED:
                            case AppUtils.ATS_STATUS_DUPLICATE_INCORRECT:
                                //case AppUtils.ATS_STATUS_SNR:
                                data.ServiceData.OutcomeStatusColor = "#FF0000"; //Red
                                break;
                            case AppUtils.ATS_STATUS_CSA_DATA_REVIEWED:
                            case AppUtils.ATS_STATUS_NURSE_CERTIFIED:
                            case AppUtils.ATS_STATUS_COMPLETE:
                            case AppUtils.ATS_STATUS_CSA_DATA_ENTERED:
                                data.ServiceData.OutcomeStatusColor = "#009900"; //Green
                                break;
                            case AppUtils.ATS_STATUS_PENDING_CLINICAL_REVIEW:
                                data.ServiceData.OutcomeStatusColor = "#FFAA00";// Orange
                                break;
                            default:
                                data.ServiceData.OutcomeStatusColor = "#FFFFFF"; //White
                                break;
                        }


                    }


                }
            }

            catch (Exception ex)
            {
                Console.WriteLine("An exception captured " + ex.Message);
            }
        }

        public bool SelectedUserChanged(object propertyPath)
        {
            bool result = false;
            try
            {
                var data = propertyPath as ObjectBase;

                DateTime time = DateTime.Now.ToLocalTime();
                DateTime EndTime = DateTime.Parse(data.LastName);
                DateTime StartTime = DateTime.Parse(data.FirstName);
                if (time < StartTime || time > EndTime)
                {
                    dynamic checkdata = this.Data;
                    TaskScreenArgs args = new TaskScreenArgs()
                    {
                        TaskTo = data.Text ?? "",
                        UserName = Utilities.Common.UserName ?? "",
                        TaskName = checkdata.NextTask.Task.TaskName ?? "",
                        Contract = checkdata.ScreenArgs.Contract ?? "",
                        AuthCPTCode = checkdata.ScreenArgs.AuthCPTCode ?? checkdata.AuthHeaderDetails.ProcedureCode.Text ?? ""
                    };
                    var output = this.Model.IsValidQueue(args);
                    List<string> _queuelist = new List<string>();
                    _queuelist = output.Text.Split(',').ToList<string>();
                    MemberSnapshotModel _model = new MemberSnapshotModel();
                    RequestBase temp = new RequestBase()
                    {
                        UserID = Utilities.Common.UserName ?? "",
                    };
                    var res = _model.UserQChangeaccess(temp);
                    try
                    {
                        if (res?.ID != "Y")
                        {
                            if (!(_queuelist.Contains(Utilities.Common.UserName)))
                            {
                                this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Alert, "You are trying to assign to a user outside of their work shift time.");
                                result = true;
                            }
                        }
                        else { }
                    }
                    catch { }
                }
                else
                {

                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("An exception captured " + ex.Message);
            }
            return result;
        }

        #region Outcome Header Validation
        private bool OutcomeHeaderValidation()
        {
            bool validate = true;
            if (this.Data != null && this.Data is OutcomeBase)
            {
                OutcomeBase data = this.Data as OutcomeBase;
                if ((data?.ScreenArgs?.TaskName ?? "") != "Long Term Outcome")
                {
                    if ((this.SelectedCycleType?.Text ?? "") == "")
                    {
                        this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Alert, "Cycle Type is missing."); return validate = false;
                    }

                    RequestBase args12 = new RequestBase()
                    {
                        Contract = this.Contract ?? "",
                        MemberID = this.MemberId ?? "",
                        UserID = Utilities.Common.UserName ?? ""
                    };

                    PremierRXModel _model = new PremierRXModel();
                    var res2 = _model.GetMemberEligibilityValidation(args12);
                    if (res2.ID == "Y" && data.ServiceData.OutcomeStatus == "Nurse Certified")
                    {
                        this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Alert, res2.Text); return validate = false;
                    }

                    if (this.Data != null && this.Data is OutcomeBase)
                    {
                        //OutcomeBase data = this.Data as OutcomeBase;
                        if ((data?.ServiceData?.OutcomeStatus ?? "") == "")
                        {
                            this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Alert, "Outcome status is missing."); return validate = false;
                        }
                        if (((data?.ServiceData?.OutcomeStatus?.ToString() ?? "") == "Missing Information") || ((data?.ServiceData?.OutcomeStatus?.ToString() ?? "") == "Pending Clinical Review"))
                        {
                            if (data.TaskDetails.Reasons.Count == 0)
                            {
                                this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Alert, "Please choose reasons."); return validate = false;
                            }
                        }
                    }
                }
            }
            try
            {
                OutcomeBase data = this.Data as OutcomeBase;
                RequestBase args2 = new RequestBase()
                {
                    MemberID = this.MemberId ?? "",
                    Contract = this.Contract ?? "",
                    TaskStatus = data.ServiceData.OutcomeStatus ?? "",
                    TaskName = data?.ScreenArgs?.TaskName ?? "",
                    Selected_Date = data.TaskDetails.DateFrom ?? "",
                    UserID = Utilities.Common.UserName ?? ""
                };
                ObjectBase TermedOut = new PremierAuthModel().termedout_DateValidation(args2);
                if (TermedOut.ID == "Y")
                {
                    this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Alert, TermedOut.Text);
                    validate = false;
                    data.ServiceData.OutcomeStatus = "Pending Clinical Review";
                    return validate;
                }
                ObjectBase LegalAid = new PremierAuthModel().LegalAidLegalAid_EligVerification(args2);
                //if (LegalAid.ID == "Y")
                //{
                //    this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Alert, LegalAid.Text);
                //    validate = false;
                //    data.ServiceData.OutcomeStatus = "Pending Clinical Review";
                //    return validate;
                //}
                if (LegalAid.ID == "L")
                {
                    int Result = this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Confirmation, LegalAid?.Text, new string[] { "Proceed", "Decline" });
                    if (Result == 0)
                    {
                        WinDTO.Tasks.ForwardEmailDetails obj = new WinDTO.Tasks.ForwardEmailDetails()
                        {
                            Memberid = this.MemberId ?? "",
                            Contract = this.Contract ?? ""
                        };
                        SendEligibilityVerificationRequest objRequest = new SendEligibilityVerificationRequest(obj);
                        objRequest.ShowDialog();
                        var viewModelA1 = objRequest.DataContext as SendEligibilityVerificationRequestVM;
                        if (viewModelA1.FaxEmailData.CloselegalButton == "True")
                        {
                            validate = false;
                            data.ServiceData.OutcomeStatus = "Pending Clinical Review";
                        }
                        else
                        {
                            validate = true;
                        }
                        return validate;
                    }
                    else if (Result == 1)
                    {
                        validate = false;
                        data.ServiceData.OutcomeStatus = "Pending Clinical Review";
                        return validate;
                    }
                }
                else if (LegalAid.ID == "Y")
                {
                    this.Window.MsgWindow.Show(WinUIDTO.ATSMsgType.Alert, LegalAid.Text);
                    validate = false;
                    data.ServiceData.OutcomeStatus = "Pending Clinical Review";
                    return validate;
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("An exception captured " + ex.Message);
            }
            return validate;
        }
