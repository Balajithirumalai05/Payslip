        [NonAction]
        public void UpdateWorkFlowStatus(string AuthNo, string WorkFlow, string WorkFlowStatus, string PopReasoncode, string ServiceID,
            string UserName, string Contract, string TaskName, string OutcomeStatus, string UserRole, string NextTask, string SelectedCPTCode)
        {
            try
            {
                if (TaskName != "Long Term Outcome")
                {
                    if (SelectedCPTCode == "SNR" && UserRole == "CSA")
                    {
                        WorkFlow = "Outcome Setup";
                        WorkFlowStatus = OutcomeStatus;
                    }
                    else if (SelectedCPTCode == "SNR" && UserRole == "RN")
                    {
                        WorkFlow = "Outcome Review";
                        WorkFlowStatus = "SNR";
                    }
                    else if (TaskName == "Enter Outcome" || TaskName == "Review Online Outcome")
                    {
                        if (OutcomeStatus == "CSA Data Entered" || OutcomeStatus == "CSA Data Reviewed")
                        {
                            WorkFlow = "Outcome Review";
                            WorkFlowStatus = "Incomplete";
                        }
                        else if (OutcomeStatus == "Missing Information")
                        {
                            if (TaskName != NextTask)
                            {
                                WorkFlow = "Outcome Review";
                                WorkFlowStatus = "Missing Information";
                            }
                            else
                            {
                                WorkFlow = "Outcome Setup";
                                WorkFlowStatus = "Missing Information";
                            }
                        }
                    }
                    else
                    {
                        if (UserRole == "RN")
                        {
                            if (OutcomeStatus == "Nurse Certified")
                            {
                                WorkFlow = "Outcome Review";
                                WorkFlowStatus = "Nurse Certified";
                            }
                            else if (OutcomeStatus == "Pending Clinical Review")
                            {
                                WorkFlow = "Outcome Review";
                                WorkFlowStatus = "Pending Clinical Review";
                            }
                            else if (OutcomeStatus == "Missing Information")
                            {
                                WorkFlow = "Outcome Review";
                                WorkFlowStatus = "Missing Information";
                            }
                        }
                        else
                        {
                            WorkFlow = "Outcome Review";
                            if (OutcomeStatus == "CSA Data Entered")
                            {
                                WorkFlowStatus = "Incomplete";
                            }
                            else
                            {
                                WorkFlowStatus = OutcomeStatus;
                            }
                        }
                    }
                }

                Dictionary<string, object> args = new Dictionary<string, object>();
                args.Add("@AuthNo", AuthNo ?? "");
                args.Add("@WorkFlow", WorkFlow ?? "");
                args.Add("@WorkFlowStatus", WorkFlowStatus ?? "");
                args.Add("@ReasonString", PopReasoncode ?? "", DBNull.Value);
                args.Add("@ServiceID", ServiceID ?? "");
                args.Add("@UserName", UserName ?? "");

                ExecuteNonQuery(SP_UpdateWorkFlowStatus, Contract ?? "", args);
            }
            catch (Exception ex)
            {
                Console.WriteLine("An error occurred: " + ex.Message);
                throw;
            }
        }

        [HttpPost]
        [ValidateFilter, ExceptionFilter]
        public List<ObjectBase> GetUpcodeDowncode()
        {
            NurseCerifyInfo objNurseCerifyInfo = ATSParams?.ToJsonObject<NurseCerifyInfo>();
            Dictionary<string, object> args = new Dictionary<string, object>();
            args.Add("@CPTFamily", objNurseCerifyInfo?.CPTFamily ?? "");
            args.Add("@CPTCode", objNurseCerifyInfo?.WINCode ?? "");
            args.Add("@Type", objNurseCerifyInfo?.Type ?? "");
            args.Add("@MemberId", objNurseCerifyInfo?.MemberId ?? "");
            args.Add("@Contract", objNurseCerifyInfo?.Contract ?? "");
            args.Add("@AuthNumber", objNurseCerifyInfo?.AuthNo ?? "");
            args.Add("@IsGeneticOutcome", objNurseCerifyInfo?.Igenomixtype ?? "");
            args.Add("@EmbryosReceived", objNurseCerifyInfo?.Emborecived ?? "");

            List<ObjectBase> objresult = new List<ObjectBase>();

            try
            {
                using (DataSet ds = ExecuteDB(SP_LoadUpcodeDownCode, objNurseCerifyInfo?.Contract, args))
                {
                    if (ds != null && ds.Tables.Count > 0 && ds.Tables[0].Rows.Count > 0)
                    {
                        DataTable documentsTable = ds.Tables[0];
                        foreach (DataRow documentRows in documentsTable.Rows)
                        {
                            objresult.Add(new ObjectBase()
                            {
                                ID = documentRows.Value<string>("HCPCSCode"),
                                Text = documentRows.Value<string>("CancelCycleCode"),
                                LoginId = documentRows.Value<string>("CancelledType"),
                                Tag = documentRows.Value<string>("ServiceRate"),
                                ObjectTag = documentRows.Value<string>("WINMarginServiceRate")
                            });
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("An error occurred: " + ex.Message);
            }

            return objresult;
        }

        [HttpPost]
        [ValidateFilter, ExceptionFilter]
        public string GetDocBytes()
        {
            string result = null;
            WinDTO.Tasks.DocBytesParams input = ATSParams?.ToJsonObject<DocBytesParams>();
            try
            {
                Dictionary<string, object> args = new Dictionary<string, object>();
                args.Add("@FaxID", input?.DocID ?? "");
                using (DataSet ds = ExecuteDB(SP_GetDocBytes, input?.Contract ?? "", args))
                {
                    if (ds != null && ds.Tables.Count > 0)
                    {
                        if (ds.Tables[0].Rows.Count > 0)
                        {
                            object pdfDocumentValue = ds.Tables[0].Rows[0]["PDF_Document"];
                            if (pdfDocumentValue != DBNull.Value && pdfDocumentValue != null)
                            {
                                result = Convert.ToBase64String((byte[])pdfDocumentValue);
                            }
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"An error occurred in GetDocBytes: {ex.Message}");
            }
            return result ?? "";
        }
